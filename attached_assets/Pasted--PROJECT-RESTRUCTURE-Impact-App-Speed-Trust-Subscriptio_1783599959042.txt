# PROJECT RESTRUCTURE: Impact App - Speed, Trust & Subscription Overhaul

## 🎯 CRITICAL OBJECTIVES

### 1. ⚡ SPEED: 0-1 Second Scan Analysis
- **Barcode-first architecture**: Instant lookup for packaged items (<100ms)
- **Redis caching layer**: Cache all scanned products for instant retrieval
- **Optimized GPT-4o prompts**: Streamlined vision analysis (<3 seconds)
- **Local database**: Pre-loaded barcode database for common products
- **Target metrics**: 
  - Barcode scans: <1 second
  - Vision scans: <3 seconds
  - Cached products: <100ms

### 2. ✅ TRUST: 100% Accurate Detection
- **Barcode verification**: 100% accuracy for packaged items with barcodes
- **Confidence thresholds**: 
  - Barcode matches: 100% (VERIFIED badge)
  - Vision scans: >90% confidence required
  - Below 90%: "Review Needed" badge with manual verification option
- **Multi-database integration**:
  - Open Food Facts API
  - USDA FoodData Central
  - Local barcode database
- **Accuracy guarantees**: Show data source for every scan

### 3.  BROAD IMPACT DESCRIPTIONS
**Short-Term Effects (1-4 hours)**:
- 3-4 detailed paragraphs (minimum 300 words)
- Cover: Energy response, blood sugar, hydration, digestion, mental clarity
- Scientific backing with citations
- Personalized based on user health profile

**Medium-Term Effects (7-30 days)**:
- 3-4 detailed paragraphs (minimum 300 words)
- Cover: Energy stability, weight changes, sleep quality, mood patterns
- Trend analysis and predictions
- Habit formation insights

**Long-Term Effects (1+ years)**:
- 3-4 detailed paragraphs (minimum 300 words)
- Cover: Metabolic health, chronic disease risk, nutritional balance
- Statistical risk assessments
- Preventive health recommendations

### 4. 🆓 FREE TIER: 3 SCANS LIFETIME
- **Hard limit**: 3 scans TOTAL (not per day, not per month)
- **No reset**: Free users never get additional scans
- **Mandatory upgrade**: After 3rd scan, block further scans
- **Clear messaging**: "You've used all 3 free scans. Upgrade to continue."
- **Grace period**: None - immediate block after limit

### 5.  SUBSCRIPTION LIMITS & ENFORCEMENT
**Tier Structure**:
- **Starter**: 100 scans/month - $1.99/month or $19.99/year
- **Pro**: 500 scans/month - $4.99/month or $49.99/year  
- **Elite**: 2,000 scans/month - $9.99/month or $99.99/year
- **Family**: Unlimited scans (4 users) - $14.99/month or $149.99/year

**Limit Enforcement**:
- Real-time scan counter
- When limit reached: Block scan, show upgrade modal
- Even if month not finished: Show "Upgrade for more scans"
- Clear messaging: "You've used 100/100 scans this month"
- Reset date displayed: "Resets on [date]" or "Upgrade now for unlimited"

- Replace ALL instances of "Liquid Impact" with "Impact"
- Update: app.json, package.json, UI strings, metadata, icons

// artifacts/impact/services/api.ts

import axios, { AxiosInstance, AxiosError } from 'axios';
import * as SecureStore from 'expo-secure-store';
import { User, Scan, ScanResult, UserStats, WeeklyTrend, SubscriptionPlan } from '@/types';

const API_BASE_URL = process.env.EXPO_PUBLIC_API_URL || 'https://api.impact.app';

class ApiService {
  private client: AxiosInstance;
  private refreshTokenPromise: Promise<string> | null = null;

  constructor() {
    this.client = axios.create({
      baseURL: API_BASE_URL,
      timeout: 30000,
      headers: {
        'Content-Type': 'application/json',
      },
    });

    this.setupInterceptors();
  }

  private setupInterceptors() {
    this.client.interceptors.request.use(
      async (config) => {
        const token = await SecureStore.getItemAsync('accessToken');
        if (token) {
          config.headers.Authorization = `Bearer ${token}`;
        }
        return config;
      },
      (error) => Promise.reject(error)
    );

    this.client.interceptors.response.use(
      (response) => response,
      async (error: AxiosError) => {
        const originalRequest = error.config;

        if (error.response?.status === 401 && !originalRequest?._retry) {
          originalRequest._retry = true;

          try {
            const newToken = await this.refreshAccessToken();
            originalRequest.headers.Authorization = `Bearer ${newToken}`;
            return this.client(originalRequest);
          } catch (refreshError) {
            await SecureStore.deleteItemAsync('accessToken');
            await SecureStore.deleteItemAsync('refreshToken');
            throw refreshError;
          }
        }

        return Promise.reject(error);
      }
    );
  }

  private async refreshAccessToken(): Promise<string> {
    if (this.refreshTokenPromise) {
      return this.refreshTokenPromise;
    }

    this.refreshTokenPromise = (async () => {
      const refreshToken = await SecureStore.getItemAsync('refreshToken');
      
      if (!refreshToken) {
        throw new Error('No refresh token available');
      }

      const response = await axios.post(`${API_BASE_URL}/auth/refresh`, {
        refreshToken,
      });

      const { accessToken, refreshToken: newRefreshToken } = response.data;
      
      await SecureStore.setItemAsync('accessToken', accessToken);
      await SecureStore.setItemAsync('refreshToken', newRefreshToken);

      this.refreshTokenPromise = null;
      return accessToken;
    })();

    return this.refreshTokenPromise;
  }

  // Auth Methods
  async signUp(email: string, password: string, displayName?: string) {
    const response = await this.client.post('/auth/signup', {
      email,
      password,
      displayName,
    });
    
    const { user, accessToken, refreshToken } = response.data;
    await SecureStore.setItemAsync('accessToken', accessToken);
    await SecureStore.setItemAsync('refreshToken', refreshToken);
    
    return user as User;
  }

  async signIn(email: string, password: string) {
    const response = await this.client.post('/auth/signin', {
      email,
      password,
    });
    
    const { user, accessToken, refreshToken } = response.data;
    await SecureStore.setItemAsync('accessToken', accessToken);
    await SecureStore.setItemAsync('refreshToken', refreshToken);
    
    return user as User;
  }

  async signOut() {
    await this.client.post('/auth/signout');
    await SecureStore.deleteItemAsync('accessToken');
    await SecureStore.deleteItemAsync('refreshToken');
  }

  // Scan Methods
  async scanBarcode(barcode: string): Promise<ScanResult> {
    const response = await this.client.post<ScanResult>('/scans/barcode', {
      barcode,
      timestamp: new Date().toISOString(),
    });
    return response.data;
  }

  async scanImage(imageUri: string): Promise<ScanResult> {
    const formData = new FormData();
    formData.append('image', {
      uri: imageUri,
      type: 'image/jpeg',
      name: 'scan.jpg',
    } as any);

    const response = await this.client.post<ScanResult>('/scans/image', formData, {
      headers: {
        'Content-Type': 'multipart/form-data',
      },
    });
    return response.data;
  }

  async getScanHistory(limit: number = 20, offset: number = 0) {
    const response = await this.client.get<{ scans: Scan[]; total: number }>('/scans/history', {
      params: { limit, offset },
    });
    return response.data;
  }

  async getScanById(scanId: string) {
    const response = await this.client.get<{ scan: Scan }>(`/scans/${scanId}`);
    return response.data.scan;
  }

  // User Methods
  async getCurrentUser() {
    const response = await this.client.get<{ user: User }>('/user/me');
    return response.data.user;
  }

  async getUserStats() {
    const response = await this.client.get<UserStats>('/user/stats');
    return response.data;
  }

  async getWeeklyTrends(weeks: number = 4) {
    const response = await this.client.get<{ trends: WeeklyTrend[] }>('/user/trends', {
      params: { weeks },
    });
    return response.data.trends;
  }

  // Subscription Methods
  async getSubscriptionPlans() {
    const response = await this.client.get<{ plans: Record<string, SubscriptionPlan> }>('/subscription/plans');
    return response.data.plans;
  }

  async createCheckoutSession(tier: string, billingCycle: 'monthly' | 'yearly') {
    const response = await this.client.post<{ checkoutUrl: string }>('/subscription/checkout', {
      tier,
      billingCycle,
    });
    return response.data.checkoutUrl;
  }

  async checkScanLimit(): Promise<{ allowed: boolean; scansUsed: number; scansLimit: number; upgradeRequired: boolean }> {
    const response = await this.client.get('/subscription/check-limit');
    return response.data;
  }

  // Product Database Methods
  async searchProducts(query: string) {
    const response = await this.client.get<{ products: any[] }>('/products/search', {
      params: { query },
    });
    return response.data.products;
  }

  async getProductByBarcode(barcode: string) {
    const response = await this.client.get(`/products/barcode/${barcode}`);
    return response.data;
  }
}

export const apiService = new ApiService();
export default apiService;

// artifacts/impact/store/useAppStore.ts

import { create } from 'zustand';
import { persist, createJSONStorage } from 'zustand/middleware';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { User, Scan, SubscriptionTier, UserStats, WeeklyTrend } from '@/types';
import { apiService } from '@/services/api';

interface AppState {
  // User State
  user: User | null;
  isAuthenticated: boolean;
  isLoading: boolean;
  error: string | null;

  // Scan State
  scans: Scan[];
  currentScan: Scan | null;
  scansLoading: boolean;

  // Stats State
  userStats: UserStats | null;
  weeklyTrends: WeeklyTrend[];

  // UI State
  showScanLimitModal: boolean;
  showSubscriptionModal: boolean;
  subscriptionReason: 'limit_reached' | 'upgrade' | null;

  // Actions
  setUser: (user: User) => void;
  signOut: () => Promise<void>;
  
  // Scan Actions
  addScan: (scan: Scan) => void;
  setCurrentScan: (scan: Scan | null) => void;
  loadScanHistory: () => Promise<void>;
  
  // Stats Actions
  loadUserStats: () => Promise<void>;
  loadWeeklyTrends: () => Promise<void>;
  
  // UI Actions
  setShowScanLimitModal: (show: boolean) => void;
  setShowSubscriptionModal: (show: boolean, reason?: 'limit_reached' | 'upgrade') => void;
  
  // Check scan limit before scanning
  checkAndIncrementScan: () => Promise<{ allowed: boolean; message?: string }>;
}

export const useAppStore = create<AppState>()(
  persist(
    (set, get) => ({
      // Initial State
      user: null,
      isAuthenticated: false,
      isLoading: false,
      error: null,
      scans: [],
      currentScan: null,
      scansLoading: false,
      userStats: null,
      weeklyTrends: [],
      showScanLimitModal: false,
      showSubscriptionModal: false,
      subscriptionReason: null,

      // User Actions
      setUser: (user) => set({ user, isAuthenticated: true }),

      signOut: async () => {
        await apiService.signOut();
        set({
          user: null,
          isAuthenticated: false,
          scans: [],
          userStats: null,
          weeklyTrends: [],
        });
      },

      // Scan Actions
      addScan: (scan) => {
        set((state) => ({
          scans: [scan, ...state.scans],
          currentScan: scan,
        }));
      },

      setCurrentScan: (scan) => set({ currentScan: scan }),

      loadScanHistory: async () => {
        set({ scansLoading: true });
        try {
          const { scans } = await apiService.getScanHistory(50);
          set({ scans, scansLoading: false });
        } catch (error) {
          set({ 
            error: error instanceof Error ? error.message : 'Failed to load scans',
            scansLoading: false 
          });
        }
      },

      // Stats Actions
      loadUserStats: async () => {
        try {
          const stats = await apiService.getUserStats();
          set({ userStats: stats });
        } catch (error) {
          set({ error: error instanceof Error ? error.message : 'Failed to load stats' });
        }
      },

      loadWeeklyTrends: async () => {
        try {
          const trends = await apiService.getWeeklyTrends(4);
          set({ weeklyTrends: trends });
        } catch (error) {
          set({ error: error instanceof Error ? error.message : 'Failed to load trends' });
        }
      },

      // UI Actions
      setShowScanLimitModal: (show) => set({ showScanLimitModal: show }),

      setShowSubscriptionModal: (show, reason = null) => 
        set({ showSubscriptionModal: show, subscriptionReason: reason }),

      // Check scan limit before scanning
      checkAndIncrementScan: async () => {
        const { user } = get();
        
        if (!user) {
          return { allowed: false, message: 'Please sign in to scan' };
        }

        // Check if user has reached scan limit
        if (user.scansUsed >= user.scansLimit && user.scansLimit > 0) {
          set({ showScanLimitModal: true, showSubscriptionModal: true, subscriptionReason: 'limit_reached' });
          return { 
            allowed: false, 
            message: `You've used all ${user.scansLimit} scans. Please upgrade to continue.` 
          };
        }

        // Update scan count
        const updatedUser = {
          ...user,
          scansUsed: user.scansUsed + 1,
        };

        set({ user: updatedUser });
        return { allowed: true };
      },
    }),
    {
      name: 'impact-storage',
      storage: createJSONStorage(() => AsyncStorage),
      partialize: (state) => ({
        user: state.user,
        isAuthenticated: state.isAuthenticated,
        scans: state.scans.slice(0, 20), // Only persist last 20 scans
      }),
    }
  )
);

import React, { useState, useRef, useEffect } from 'react';
import {
  View,
  Text,
  StyleSheet,
  TouchableOpacity,
  Alert,
  ActivityIndicator,
} from 'react-native';
import { useRouter } from 'expo-router';
import { CameraView, useCameraPermissions } from 'expo-camera';
import { Ionicons } from '@expo/vector-icons';
import * as ImagePicker from 'expo-image-picker';
import { useAppStore } from '@/store/useAppStore';
import { Colors } from '@/constants/Colors';
import { useColorScheme } from '@/hooks/useColorScheme';
import { apiService } from '@/services/api';

export default function ScanScreen() {
  const router = useRouter();
  const colorScheme = useColorScheme();
  const colors = Colors[colorScheme ?? 'light'];
  const [permission, requestPermission] = useCameraPermissions();
  
  const { addScan, setCurrentScan, setShowScanLimitModal } = useAppStore();
  
  const [scanning, setScanning] = useState(true);
  const [processing, setProcessing] = useState(false);
  const scannerRef = useRef<any>(null);

  useEffect(() => {
    requestPermission();
  }, []);

  const handleBarcodeScanned = async ({ type, data }: { type: string; data: string }) => {
    if (!scanning || processing) return;
    
    setScanning(false);
    setProcessing(true);

    try {
      const result = await apiService.scanBarcode(data);
      
      if (result.upgradeRequired) {
        setShowScanLimitModal(true);
        setProcessing(false);
        return;
      }

      addScan(result.scan);
      setCurrentScan(result.scan);
      
      setTimeout(() => {
        setProcessing(false);
        router.push(`/scan/${result.scan.id}`);
      }, 500);
    } catch (error) {
      Alert.alert(
        'Scan Failed',
        error instanceof Error ? error.message : 'Unable to process barcode',
        [{ text: 'OK', onPress: () => setScanning(true) }]
      );
      setProcessing(false);
    }
  };

  const handleImagePick = async () => {
    const { status } = await ImagePicker.requestMediaLibraryPermissionsAsync();
    
    if (status !== 'granted') {
      Alert.alert('Permission Required', 'Please grant permission to access your photos');
      return;
    }

    const result = await ImagePicker.launchImageLibraryAsync({
      mediaTypes: ImagePicker.MediaTypeOptions.Images,
      allowsEditing: true,
      aspect: [3, 4],
      quality: 0.8,
    });

    if (!result.canceled && result.assets[0]) {
      setProcessing(true);
      
      try {
        const scanResult = await apiService.scanImage(result.assets[0].uri);
        
        if (scanResult.upgradeRequired) {
          setShowScanLimitModal(true);
          setProcessing(false);
          return;
        }

        addScan(scanResult.scan);
        setCurrentScan(scanResult.scan);
        
        router.push(`/scan/${scanResult.scan.id}`);
      } catch (error) {
        Alert.alert(
          'Scan Failed',
          error instanceof Error ? error.message : 'Unable to process image',
          [{ text: 'OK' }]
        );
      } finally {
        setProcessing(false);
      }
    }
  };

  if (!permission) {
    return (
      <View style={[styles.container, styles.centerContent]}>
        <ActivityIndicator size="large" color={colors.primary} />
      </View>
    );
  }

  if (!permission.granted) {
    return (
      <View style={[styles.container, styles.centerContent]}>
        <Ionicons name="camera-outline" size={64} color={colors.textSecondary} />
        <Text style={[styles.text, { color: colors.text, marginTop: 16 }]}>
          Camera permission is required to scan products
        </Text>
        <TouchableOpacity
          style={[styles.button, { backgroundColor: colors.primary }]}
          onPress={requestPermission}>
          <Text style={styles.buttonText}>Grant Permission</Text>
        </TouchableOpacity>
      </View>
    );
  }

  return (
    <View style={styles.container}>
      {processing ? (
        <View style={[styles.container, styles.centerContent, { backgroundColor: colors.background }]}>
          <ActivityIndicator size="large" color={colors.primary} />
          <Text style={[styles.text, { color: colors.text, marginTop: 16 }]}>
            Analyzing product...
          </Text>
        </View>
      ) : (
        <>
          <CameraView
            ref={scannerRef}
            style={styles.camera}
            facing="back"
            barcodeScannerSettings={{
              barcodeTypes: ['qr', 'ean13', 'ean8', 'upc_e', 'upc_a'],
            }}
            onBarcodeScanned={scanning ? handleBarcodeScanned : undefined}>
            <View style={styles.overlay}>
              <View style={styles.scanFrame}>
                <View style={[styles.corner, styles.topLeft]} />
                <View style={[styles.corner, styles.topRight]} />
                <View style={[styles.corner, styles.bottomLeft]} />
                <View style={[styles.corner, styles.bottomRight]} />
              </View>
              <Text style={styles.scanInstruction}>
                Position barcode or product in frame
              </Text>
            </View>
          </CameraView>

          <View style={styles.controls}>
            <TouchableOpacity
              style={[styles.controlButton, { backgroundColor: colors.card }]}
              onPress={handleImagePick}>
              <Ionicons name="images" size={24} color={colors.primary} />
              <Text style={[styles.controlText, { color: colors.text }]}>
                Choose Photo
              </Text>
            </TouchableOpacity>

            <TouchableOpacity
              style={[styles.controlButton, { backgroundColor: colors.card }]}
              onPress={() => setScanning(!scanning)}>
              <Ionicons 
                name={scanning ? 'pause' : 'play'} 
                size={24} 
                color={colors.primary} 
              />
              <Text style={[styles.controlText, { color: colors.text }]}>
                {scanning ? 'Pause' : 'Resume'}
              </Text>
            </TouchableOpacity>

            <TouchableOpacity
              style={[styles.controlButton, { backgroundColor: colors.card }]}
              onPress={() => router.back()}>
              <Ionicons name="close" size={24} color={colors.textSecondary} />
              <Text style={[styles.controlText, { color: colors.textSecondary }]}>
                Cancel
              </Text>
            </TouchableOpacity>
          </View>
        </>
      )}
    </View>
  );
}

```typescript
// artifacts/impact-backend/src/services/redis.ts

import Redis from 'ioredis';

class RedisService {
  private client: Redis;
  private readonly SCAN_TTL = 86400 * 30; // 30 days
  private readonly PRODUCT_TTL = 86400 * 7; // 7 days
  private readonly USER_STATS_TTL = 300; // 5 minutes

  constructor() {
    this.client = new Redis(process.env.REDIS_URL || 'redis://localhost:6379');
    
    this.client.on('error', (err) => {
      console.error('Redis error:', err);
    });
    
    this.client.on('connect', () => {
      console.log('Redis connected');
    });
  }

  // Scan caching
  async cacheScan(barcode: string, scanData: any): Promise<void> {
    const key = `scan:${barcode}`;
    await this.client.setex(key, this.SCAN_TTL, JSON.stringify(scanData));
  }

  async getCachedScan(barcode: string): Promise<any | null> {
    const key = `scan:${barcode}`;
    const data = await this.client.get(key);
    return data ? JSON.parse(data) : null;
  }

  // Product caching
  async cacheProduct(barcode: string, productData: any): Promise<void> {
    const key = `product:${barcode}`;
    await this.client.setex(key, this.PRODUCT_TTL, JSON.stringify(productData));
  }

  async getCachedProduct(barcode: string): Promise<any | null> {
    const key = `product:${barcode}`;
    const data = await this.client.get(key);
    return data ? JSON.parse(data) : null;
  }

  // User stats caching
  async cacheUserStats(userId: string, stats: any): Promise<void> {
    const key = `user:stats:${userId}`;
    await this.client.setex(key, this.USER_STATS_TTL, JSON.stringify(stats));
  }

  async getCachedUserStats(userId: string): Promise<any | null> {
    const key = `user:stats:${userId}`;
    const data = await this.client.get(key);
    return data ? JSON.parse(data) : null;
  }

  async invalidateUserStats(userId: string): Promise<void> {
    const key = `user:stats:${userId}`;
    await this.client.del(key);
  }

  // Rate limiting
  async incrementScanCount(userId: string): Promise<number> {
    const key = `user:scans:${userId}:${this.getCurrentMonth()}`;
    const count = await this.client.incr(key);
    await this.client.expire(key, 86400 * 31); // Expire after month
    return count;
  }

  async getScanCount(userId: string): Promise<number> {
    const key = `user:scans:${userId}:${this.getCurrentMonth()}`;
    const count = await this.client.get(key);
    return count ? parseInt(count) : 0;
  }

  private getCurrentMonth(): string {
    const now = new Date();
    return `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`;
  }

  // Health check
  async healthCheck(): Promise<boolean> {
    try {
      await this.client.ping();
      return true;
    } catch {
      return false;
    }
  }

  async disconnect(): Promise<void> {
    await this.client.quit();
  }
}

export const redisService = new RedisService();
export default redisService;

// artifacts/impact-backend/src/services/barcodeLookup.ts

import axios from 'axios';
import redisService from './redis';
import { prisma } from '../lib/prisma';

interface ProductData {
  barcode: string;
  name: string;
  brand: string | null;
  category: string;
  ingredients: any[];
  nutrition: {
    calories: number;
    sugarGrams: number;
    caffeineMg: number;
    sodiumMg: number;
    fatGrams: number;
    proteinGrams: number;
    carbsGrams: number;
    fiberGrams: number;
  };
  additives: string[];
  artificialSweeteners: boolean;
}

class BarcodeLookupService {
  private readonly OPEN_FOOD_FACTS_URL = 'https://world.openfoodfacts.org/api/v0/product';
  private readonly USDA_API_URL = 'https://api.nal.usda.gov/fdc/v1';
  private readonly USDA_API_KEY = process.env.USDA_API_KEY || '';

  async lookupByBarcode(barcode: string): Promise<ProductData | null> {
    // Check Redis cache first (0-100ms target)
    const cached = await redisService.getCachedProduct(barcode);
    if (cached) {
      console.log(`[Barcode] Cache hit for ${barcode}`);
      return cached;
    }

    console.log(`[Barcode] Cache miss for ${barcode}, looking up...`);

    try {
      // Try Open Food Facts first
      const offProduct = await this.lookupOpenFoodFacts(barcode);
      if (offProduct) {
        await redisService.cacheProduct(barcode, offProduct);
        await this.saveToDatabase(barcode, offProduct);
        return offProduct;
      }

      // Try USDA FoodData Central
      const usdaProduct = await this.lookupUSDA(barcode);
      if (usdaProduct) {
        await redisService.cacheProduct(barcode, usdaProduct);
        await this.saveToDatabase(barcode, usdaProduct);
        return usdaProduct;
      }

      return null;
    } catch (error) {
      console.error(`[Barcode] Lookup error for ${barcode}:`, error);
      return null;
    }
  }

  private async lookupOpenFoodFacts(barcode: string): Promise<ProductData | null> {
    try {
      const response = await axios.get(`${this.OPEN_FOOD_FACTS_URL}/${barcode}.json`, {
        timeout: 3000, // 3 second timeout
        headers: {
          'User-Agent': 'Impact App - health@impact.app',
        },
      });

      if (response.data.status !== 1) {
        return null;
      }

      const product = response.data.product;
      
      return {
        barcode,
        name: product.product_name || 'Unknown Product',
        brand: product.brands?.split(',')[0] || null,
        category: this.mapCategory(product.categories_tags?.[0]),
        ingredients: product.ingredients || [],
        nutrition: {
          calories: product.nutriments?.['energy-kcal_100g'] || 0,
          sugarGrams: product.nutriments?.sugars_100g || 0,
          caffeineMg: 0, // OFF doesn't track caffeine
          sodiumMg: (product.nutriments?.sodium_100g || 0) * 1000,
          fatGrams: product.nutriments?.fat_100g || 0,
          proteinGrams: product.nutriments?.proteins_100g || 0,
          carbsGrams: product.nutriments?.carbohydrates_100g || 0,
          fiberGrams: product.nutriments?.fiber_100g || 0,
        },
        additives: product.additives_tags || [],
        artificialSweeteners: this.detectArtificialSweeteners(product.ingredients_tags || []),
      };
    } catch (error) {
      console.error('[OpenFoodFacts] Error:', error.message);
      return null;
    }
  }

  private async lookupUSDA(barcode: string): Promise<ProductData | null> {
    try {
      const response = await axios.get(`${this.USDA_API_URL}/foods/list?api_key=${this.USDA_API_KEY}&dataType=Foundation,Survey%20%28FNDDS%29&pageSize=1`, {
        timeout: 3000,
      });

      if (!response.data.foods || response.data.foods.length === 0) {
        return null;
      }

      const food = response.data.foods[0];
      
      return {
        barcode,
        name: food.description,
        brand: null,
        category: 'other',
        ingredients: [],
        nutrition: {
          calories: food.foodNutrients?.find(n => n.nutrientName === 'Energy')?.value || 0,
          sugarGrams: food.foodNutrients?.find(n => n.nutrientName === 'Sugars, total including NLEA')?.value || 0,
          caffeineMg: food.foodNutrients?.find(n => n.nutrientName === 'Caffeine')?.value || 0,
          sodiumMg: food.foodNutrients?.find(n => n.nutrientName === 'Sodium, Na')?.value || 0,
          fatGrams: food.foodNutrients?.find(n => n.nutrientName === 'Total lipid (fat)')?.value || 0,
          proteinGrams: food.foodNutrients?.find(n => n.nutrientName === 'Protein')?.value || 0,
          carbsGrams: food.foodNutrients?.find(n => n.nutrientName === 'Carbohydrate, by difference')?.value || 0,
          fiberGrams: food.foodNutrients?.find(n => n.nutrientName === 'Fiber, total dietary')?.value || 0,
        },
        additives: [],
        artificialSweeteners: false,
      };
    } catch (error) {
      console.error('[USDA] Error:', error.message);
      return null;
    }
  }

  private mapCategory(categoryTag?: string): string {
    if (!categoryTag) return 'other';
    
    const categoryMap: Record<string, string> = {
      'en:waters': 'water',
      'en:fruit-juices': 'juice',
      'en:carbonated-drinks': 'soda',
      'en:energy-drinks': 'energy_drink',
      'en:coffees': 'coffee',
      'en:teas': 'tea',
      'en:milks': 'milk',
      'en:plant-milks': 'plant_milk',
      'en:smoothies': 'smoothie',
      'en:sports-drinks': 'sports_drink',
      'en:alcoholic-beverages': 'alcohol',
    };

    return categoryMap[categoryTag] || 'other';
  }

  private detectArtificialSweeteners(ingredients: string[]): boolean {
    const sweeteners = [
      'aspartame',
      'sucralose',
      'saccharin',
      'acesulfame',
      'neotame',
      'advantame',
    ];
    
    return ingredients.some(ing => 
      sweeteners.some(sweetener => ing.toLowerCase().includes(sweetener))
    );
  }

  private async saveToDatabase(barcode: string, product: ProductData): Promise<void> {
    try {
      await prisma.product.upsert({
        where: { barcode },
        update: {
          name: product.name,
          brand: product.brand,
          category: product.category as any,
          ...product.nutrition,
          ingredients: product.ingredients,
          additives: product.additives,
          artificialSweeteners: product.artificialSweeteners,
          dataSource: 'openfoodfacts',
          verified: true,
          confidence: 1.0,
          updatedAt: new Date(),
        },
        create: {
          barcode,
          name: product.name,
          brand: product.brand,
          category: product.category as any,
          ...product.nutrition,
          ingredients: product.ingredients,
          additives: product.additives,
          artificialSweeteners: product.artificialSweeteners,
          dataSource: 'openfoodfacts',
          verified: true,
          confidence: 1.0,
        },
      });
    } catch (error) {
      console.error('[Database] Save error:', error);
    }
  }
}

// artifacts/impact-backend/src/index.ts

import express, { Express } from 'express';
import cors from 'cors';
import helmet from 'helmet';
import rateLimit from 'express-rate-limit';
import { prisma } from './lib/prisma';
import { redisService } from './services/redis';
import authRoutes from './routes/auth';
import scanRoutes from './routes/scans';
import userRoutes from './routes/user';
import subscriptionRoutes from './routes/subscription';
import productRoutes from './routes/products';
import { errorHandler } from './middleware/errorHandler';
import { logger } from './lib/logger';

const app: Express = express();
const PORT = process.env.PORT || 3000;

// Security middleware
app.use(helmet());
app.use(cors({
  origin: process.env.CORS_ORIGIN || '*',
  credentials: true,
}));

// Rate limiting
const limiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15 minutes
  max: 100, // limit each IP to 100 requests per windowMs
  message: 'Too many requests from this IP, please try again later.',
});
app.use('/api/', limiter);

// Body parsing
app.use(express.json({ limit: '10mb' }));
app.use(express.urlencoded({ extended: true, limit: '10mb' }));

// Request logging
app.use((req, res, next) => {
  logger.info({
    method: req.method,
    url: req.url,
    ip: req.ip,
    userAgent: req.get('user-agent'),
  }, 'Request received');
  next();
});

// Routes
app.use('/api/auth', authRoutes);
app.use('/api/scans', scanRoutes);
app.use('/api/user', userRoutes);
app.use('/api/subscription', subscriptionRoutes);
app.use('/api/products', productRoutes);

// Health check
app.get('/api/health', async (req, res) => {
  const dbHealthy = await prisma.$queryRaw`SELECT 1`.then(() => true).catch(() => false);
  const redisHealthy = await redisService.healthCheck();
  
  const healthy = dbHealthy && redisHealthy;
  
  res.status(healthy ? 200 : 503).json({
    status: healthy ? 'healthy' : 'unhealthy',
    timestamp: new Date().toISOString(),
    services: {
      database: dbHealthy ? 'connected' : 'disconnected',
      redis: redisHealthy ? 'connected' : 'disconnected',
    },
  });
});

// Error handling
app.use(errorHandler);

// Start server
app.listen(PORT, async () => {
  logger.info({ port: PORT }, `Impact API server running on port ${PORT}`);
  
  // Test database connection
  try {
    await prisma.$connect();
    logger.info('Database connected');
  } catch (error) {
    logger.error({ error }, 'Database connection failed');
    process.exit(1);
  }
});

// Graceful shutdown
process.on('SIGTERM', async () => {
  logger.info('SIGTERM received, shutting down gracefully...');
  
  await prisma.$disconnect();
  await redisService.disconnect();
  
  process.exit(0);
});

export default app;
```

// artifacts/impact-backend/src/routes/scans.ts

import { Router, Request, Response } from 'express';
import { authenticate } from '../middleware/auth';
import { barcodeLookupService } from '../services/barcodeLookup';
import { aiAnalysisService } from '../services/aiAnalysis';
import { prisma } from '../lib/prisma';
import { redisService } from '../services/redis';
import { z } from 'zod';

const router = Router();

// Scan barcode
router.post('/barcode', authenticate, async (req: Request, res: Response) => {
  try {
    const { barcode } = z.object({
      barcode: z.string().min(8).max(13),
    }).parse(req.body);

    const userId = req.user!.id;

    // Check scan limit
    const user = await prisma.user.findUnique({
      where: { id: userId },
      select: { scansUsed: true, scansLimit: true, subscriptionTier: true },
    });

    if (!user) {
      return res.status(404).json({ error: 'User not found' });
    }

    if (user.scansUsed >= user.scansLimit && user.scansLimit > 0) {
      return res.status(403).json({
        error: 'Scan limit reached',
        message: `You've used all ${user.scansLimit} scans. Please upgrade to continue.`,
        upgradeRequired: true,
      });
    }

    // Check Redis cache first (target: <100ms)
    const cachedResult = await redisService.getCachedScan(barcode);
    if (cachedResult) {
      // Increment scan count
      await prisma.user.update({
        where: { id: userId },
        data: { scansUsed: { increment: 1 } },
      });

      return res.json({
        scan: cachedResult,
        verified: true,
        confidenceMessage: 'Product verified from barcode database',
        scansRemaining: user.scansLimit - user.scansUsed - 1,
      });
    }

    // Lookup product from barcode databases (target: <1s)
    const productData = await barcodeLookupService.lookupByBarcode(barcode);

    if (!productData) {
      return res.status(404).json({
        error: 'Product not found',
        message: 'Unable to find product information for this barcode. Please try scanning the product image instead.',
      });
    }

    // Generate AI analysis (target: <2s)
    const analysis = await aiAnalysisService.analyzeProduct({
      productName: productData.name,
      brand: productData.brand,
      category: productData.category as any,
      nutrition: productData.nutrition,
      ingredients: productData.ingredients,
      additives: productData.additives,
      artificialSweeteners: productData.artificialSweeteners,
    });

    // Create scan record
    const scan = await prisma.scan.create({
      data: {
        userId,
        barcode,
        productId: null, // Will be linked if product exists
        imageUrl: '',
        detectedProduct: productData.name,
        brand: productData.brand,
        category: productData.category as any,
        confidenceScore: 1.0,
        impactScore: analysis.impactScore,
        hydrationLevel: analysis.hydrationLevel,
        glycemicImpact: analysis.glycemicImpact,
        verified: true,
        dataSource: 'BARCODE',
        
        // Impact descriptions
        shortTermPara1: analysis.shortTermEffects.para1,
        shortTermPara2: analysis.shortTermEffects.para2,
        shortTermPara3: analysis.shortTermEffects.para3,
        shortTermPara4: analysis.shortTermEffects.para4,
        shortTermWordCount: analysis.shortTermEffects.wordCount,
        
        mediumTermPara1: analysis.mediumTermEffects.para1,
        mediumTermPara2: analysis.mediumTermEffects.para2,
        mediumTermPara3: analysis.mediumTermEffects.para3,
        mediumTermPara4: analysis.mediumTermEffects.para4,
        mediumTermWordCount: analysis.mediumTermEffects.wordCount,
        
        longTermPara1: analysis.longTermEffects.para1,
        longTermPara2: analysis.longTermEffects.para2,
        longTermPara3: analysis.longTermEffects.para3,
        longTermPara4: analysis.longTermEffects.para4,
        longTermWordCount: analysis.longTermEffects.wordCount,
        
        // Composition
        servingSize: 100,
        servingUnit: 'g',
        ...productData.nutrition,
        additives: productData.additives,
        artificialSweeteners: productData.artificialSweeteners,
      },
    });

    // Cache result for future scans
    await redisService.cacheScan(barcode, scan);

    // Increment scan count
    await prisma.user.update({
      where: { id: userId },
      data: { scansUsed: { increment: 1 } },
    });

    res.json({
      scan,
      verified: true,
      confidenceMessage: 'Product verified from barcode database',
      scansRemaining: user.scansLimit - user.scansUsed - 1,
    });

  } catch (error) {
    if (error instanceof z.ZodError) {
      return res.status(400).json({ error: 'Invalid barcode format' });
    }
    
    console.error('[Barcode Scan] Error:', error);
    res.status(500).json({ error: 'Failed to process barcode scan' });
  }
});

// Scan image (vision-based)
router.post('/image', authenticate, async (req: Request, res: Response) => {
  // Implementation for image-based scanning
  // This would use GPT-4 Vision or similar
  res.status(501).json({ error: 'Image scanning not yet implemented' });
});

// Get scan history
router.get('/history', authenticate, async (req: Request, res: Response) => {
  try {
    const userId = req.user!.id;
    
    const { limit = 20, offset = 0 } = req.query;
    
    const [scans, total] = await Promise.all([
      prisma.scan.findMany({
        where: { userId },
        orderBy: { createdAt: 'desc' },
        take: Number(limit),
        skip: Number(offset),
      }),
      prisma.scan.count({ where: { userId } }),
    ]);

    res.json({ scans, total });
  } catch (error) {
    console.error('[Scan History] Error:', error);
    res.status(500).json({ error: 'Failed to fetch scan history' });
  }
});

// Get single scan
router.get('/:id', authenticate, async (req: Request, res: Response) => {
  try {
    const { id } = req.params;
    const userId = req.user!.id;

    const scan = await prisma.scan.findFirst({
      where: { id, userId },
    });

    if (!scan) {
      return res.status(404).json({ error: 'Scan not found' });
    }

    res.json({ scan });
  } catch (error) {
    console.error('[Get Scan] Error:', error);
    res.status(500).json({ error: 'Failed to fetch scan' });
  }
});

export default router;
