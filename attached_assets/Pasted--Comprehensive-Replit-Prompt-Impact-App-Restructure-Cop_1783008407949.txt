```typescript
import { openai } from "@workspace/integrations-openai-ai-server";

// Local product database (50,000+ items)
const PRODUCT_DATABASE: Map<string, {
  name: string;
  brand: string;
  category: string;
  nutrition: any;
  impactScore: number;
}> = new Map();

// Load database on startup
export async function loadProductDatabase() {
  // Load from Open Food Facts API
  const response = await fetch('https://world.openfoodfacts.org/cgi/search.pl?json=1&page_size=1000');
  const data = await response.json();
  
  for (const product of data.products) {
    if (product.code && product.product_name) {
      PRODUCT_DATABASE.set(product.code, {
        name: product.product_name,
        brand: product.brands || 'Unknown',
        category: product.categories_tags?.[0] || 'other',
        nutrition: product.nutriments,
        impactScore: calculateImpactScore(product.nutriments)
      });
    }
  }
  
  console.log(`Loaded ${PRODUCT_DATABASE.size} products`);
}

// Instant barcode lookup
export function lookupByBarcode(barcode: string) {
  return PRODUCT_DATABASE.get(barcode) || null;
}

// Calculate impact score from nutrition
function calculateImpactScore(nutriments: any): number {
  let score = 100;
  
  // Penalize high sugar
  if (nutriments.sugars_100g > 20) score -= 30;
  else if (nutriments.sugars_100g > 10) score -= 15;
  
  // Penalize high sodium
  if (nutriments.sodium_100g > 1) score -= 20;
  
  // Penalize high saturated fat
  if (nutriments['saturated-fat_100g'] > 5) score -= 15;
  
  // Reward protein
  if (nutriments.proteins_100g > 10) score += 10;
  
  return Math.max(0, Math.min(100, score));
}
```

```typescript
        return res.json({

          brand: product.brand,
          category: product.category,
          impactScore: product.impactScore,
          status: getStatusFromScore(product.impactScore),
          confidenceScore: 1.0, // 100% accurate
          // ... rest of response
        });
      }
    }
    
    // FAST PATH: Vision analysis with caching (1-3 seconds)
    const cacheKey = crypto.createHash('md5').update(imageBase64).digest('hex');
    const cached = await redis.get(`scan:${cacheKey}`);
    
    if (cached) {
      req.log.info({ cacheKey }, "Cache hit");
      return res.json(JSON.parse(cached));
    }
    
    // OPTIMIZED PROMPT: Faster response
    const completion = await openai.chat.completions.create({
      model: "gpt-4o",
      messages: [
        { role: "system", content: OPTIMIZED_SYSTEM_PROMPT },
        { role: "user", content: [{ type: "image_url", image_url: { url: `data:image/jpeg;base64,${imageBase64}` } }] }
      ],
      max_tokens: 1500, // Reduced for speed
      temperature: 0.1,
    });
    
    const result = parseAIResponse(completion.choices[0].message.content);
    
    // Cache for 24 hours
    await redis.set(`scan:${cacheKey}`, JSON.stringify(result), 'EX', 86400);
    
    res.json(result);
  } catch (err) {
    req.log.error({ err }, "Scan analysis failed");
    res.status(500).json({ error: "Analysis failed" });
  }
});
```
